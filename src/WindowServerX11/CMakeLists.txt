# The X11 side of the WindowServer port (M3).
#
#   X11Context.*         -- shared X display + window + store; reads X events and
#                           delivers input via the neutral bridge below
#   X11ScreenBackend.*   -- ScreenBackend impl (display) + factory hook
#   X11InputMap.cpp      -- Serenity-only mapper: RawX11Event -> KeyEvent / MousePacket
#   X11InputBackend.h    -- neutral bridge types (no Xlib, no Serenity KeyCode)
#
# Two TUs only, because Xlib's `KeyCode` typedef collides with Serenity's `enum
# KeyCode`: X11Context.* includes Xlib (not Serenity's KeyCode), X11InputMap.cpp
# includes Serenity (not Xlib). They meet at the neutral types in the header.
# Compiled into a static library that a tracked patch links into WindowServer.

find_package(PkgConfig REQUIRED)
pkg_check_modules(X11_PC REQUIRED IMPORTED_TARGET x11)
pkg_check_modules(XEXT_PC REQUIRED IMPORTED_TARGET xext)

add_library(serenade_x11 STATIC
    X11Context.cpp
    X11ScreenBackend.cpp
    X11InputMap.cpp
)

# Compile against the Serenity WindowServer headers (the backend subclasses
# WindowServer::ScreenBackend) plus the host sys/devices/gpu.h shim generated by
# the WindowServer target during configure.
target_include_directories(serenade_x11 PRIVATE
    ${SERENITY_SOURCE_DIR}
    ${SERENITY_SOURCE_DIR}/Userland
    ${SERENITY_SOURCE_DIR}/Userland/Libraries
    ${SERENITY_SOURCE_DIR}/Userland/Services
    ${CMAKE_BINARY_DIR}
    ${CMAKE_BINARY_DIR}/lagom
    ${CMAKE_BINARY_DIR}/lagom/Userland/Libraries
    ${CMAKE_BINARY_DIR}/lagom/Userland/Services/WindowServer/include
)

target_link_libraries(serenade_x11 PRIVATE AK LibCore LibGfx PkgConfig::X11_PC PkgConfig::XEXT_PC)

# Header-only input seam consumed by tests/ (kept as an interface alias).
add_library(serenade_input_backend INTERFACE)
target_include_directories(serenade_input_backend INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
